   kubectl -n {{ .Release.Namespace }} get svc {{ include "banking-web-application.fullname" . }} -w
{{- else }}
   kubectl -n {{ .Release.Namespace }} port-forward svc/{{ include "banking-web-application.fullname" . }} {{ .Values.service.port }}:{{ .Values.service.port }}

   http://localhost:{{ .Values.service.port }}
{{- end }}
{{- if .Values.ingress.enabled }}

   Ingress is enabled:  http://{{ .Values.ingress.host }}
   (add "127.0.0.1 {{ .Values.ingress.host }}" to /etc/hosts for a local cluster)
{{- end }}

3. IMPORTANT: `legacy.serviceUrl` must be an address your BROWSER can reach -
   this app's dashboard calls legacy-service directly with client-side
   fetch(), it does not proxy through this pod. An in-cluster DNS name
   (like {{ include "banking-web-application.fullname" . }}-style
   *.svc.cluster.local addresses) will NOT work here. Point it at
   legacy-service's own NodePort, Ingress host, or LoadBalancer address.

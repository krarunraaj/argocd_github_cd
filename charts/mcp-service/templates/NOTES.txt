{{- else }}
   kubectl -n {{ .Release.Namespace }} port-forward svc/{{ include "mcp-service.fullname" . }} {{ .Values.service.port }}:{{ .Values.service.port }}
   Then:              http://localhost:{{ .Values.service.port }}/mcp
{{- end }}

3. Validate:

   curl -s $baseUrl/health
   curl -s $baseUrl/ready          # "legacy": "up" means the boundary works

   # the reference MCP client
   cd client && MCP_SERVICE_URL=$baseUrl uv run --project ../mcp-service python call_mcp.py CIF908123

   # or the Postman suite (full initialize handshake)
   newman run deploy/postman/mcp-service.postman_collection.json \
     -e deploy/postman/{{ if eq .Values.service.type "NodePort" }}nodeport{{ else }}portforward{{ end }}.postman_environment.json

NOTE: this release exposes the MCP endpoint without inbound authentication.
      That is a training convenience ({{ .Values.service.type }} + no auth).
